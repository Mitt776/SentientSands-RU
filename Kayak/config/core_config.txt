# ─── Retrieval limits ─────────────────────────────────────────────────────────

# How many keywords to extract from the player message (Layer 0 seeds)

# How many layer-expansion rounds to run after the initial keyword match
# Layer 0 = direct match only.  Layer 1 = expand from matched entities. Etc.
max_layers = 1

# Per-entity: how many field-value references to follow per expansion round

# Absolute ceiling on how many entities end up in the prompt context
# (Priority entities like target_npc are on top of this limit)
max_files = 4

# Hard timeout in milliseconds for the entire retrieval cycle
timeout_ms = 500

# ─── Dialogue ─────────────────────────────────────────────────────────────────

# How many lines of dialogue.txt to keep in the active file.
# Lines beyond this are archived to IGN_dialogue_backup/archive.txt.

# How many dialogue history lines Kayak injects into normal chat prompts.
# This can be higher than active storage only if dialogue was preserved externally.
prompt_dialogue_keep_lines = 40

# How many dialogue lines Kayak injects for yell/group chat prompts.
# Lower helps keep crowd prompts under budget.
yell_dialogue_keep_lines = 20

# ─── Non-expandable fields ────────────────────────────────────────────────────
# Structural fields that are NEVER used as world-reference seeds.

non_expand_fields = id, persistent_id, runtime_id, weight, display_name, aliases, average_price, base_price_cats, price_modifier
